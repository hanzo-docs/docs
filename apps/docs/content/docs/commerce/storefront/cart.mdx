This guide covers cart creation, item management, and coupon application in your storefront.

## Cart Context

Create a React context to manage cart state across your app:

```tsx
'use client'

import { createContext, useContext, useState, useEffect } from 'react'
import { commerce } from '@/lib/commerce'
import type { Cart } from '@hanzo/commerce'

const CartContext = createContext<{
  cart: Cart | null
  addItem: (variantId: string, quantity?: number) => Promise<void>
  removeItem: (itemId: string) => Promise<void>
  updateQuantity: (itemId: string, quantity: number) => Promise<void>
  applyCoupon: (code: string) => Promise<void>
}>({} as any)

export function CartProvider({ children, regionId }: {
  children: React.ReactNode
  regionId: string
}) {
  const [cart, setCart] = useState<Cart | null>(null)

  useEffect(() => {
    const cartId = localStorage.getItem('cartId')
    if (cartId) {
      commerce.carts.get(cartId).then(setCart).catch(() => {
        localStorage.removeItem('cartId')
      })
    }
  }, [])

  async function getOrCreateCart() {
    if (cart) return cart
    const newCart = await commerce.carts.create({ regionId })
    localStorage.setItem('cartId', newCart.id)
    setCart(newCart)
    return newCart
  }

  async function addItem(variantId: string, quantity = 1) {
    const c = await getOrCreateCart()
    const updated = await commerce.carts.addItem(c.id, { variantId, quantity })
    setCart(updated)
  }

  async function removeItem(itemId: string) {
    if (!cart) return
    const updated = await commerce.carts.removeItem(cart.id, itemId)
    setCart(updated)
  }

  async function updateQuantity(itemId: string, quantity: number) {
    if (!cart) return
    const updated = await commerce.carts.updateItem(cart.id, itemId, { quantity })
    setCart(updated)
  }

  async function applyCoupon(code: string) {
    if (!cart) return
    const updated = await commerce.carts.applyPromotion(cart.id, { code })
    setCart(updated)
  }

  return (
    <CartContext.Provider value={{ cart, addItem, removeItem, updateQuantity, applyCoupon }}>
      {children}
    </CartContext.Provider>
  )
}

export const useCart = () => useContext(CartContext)
```

## Add to Cart Button

```tsx
'use client'

import { useCart } from '@/components/cart-provider'
import { Button } from '@hanzo/gui'

export function AddToCartButton({ variantId }: { variantId: string }) {
  const { addItem } = useCart()
  const [loading, setLoading] = useState(false)

  async function handleClick() {
    setLoading(true)
    await addItem(variantId)
    setLoading(false)
  }

  return (
    <Button onPress={handleClick} disabled={loading}>
      {loading ? 'Adding...' : 'Add to Cart'}
    </Button>
  )
}
```

## Cart Drawer

```tsx
'use client'

import { useCart } from '@/components/cart-provider'
import { Button, Input, Paragraph, Separator, XStack, YStack } from '@hanzo/gui'

export function CartDrawer() {
  const { cart, removeItem, updateQuantity } = useCart()
  if (!cart || cart.items.length === 0) return <Paragraph>Your cart is empty</Paragraph>

  return (
    <YStack>
      {cart.items.map(item => (
        <XStack key={item.id} justifyContent="space-between" paddingVertical="$4" gap="$4">
          <YStack>
            <Paragraph fontWeight="500">{item.productName}</Paragraph>
            <Paragraph size="$2" theme="alt2">
              ${(item.unitPrice / 100).toFixed(2)} x {item.quantity}
            </Paragraph>
          </YStack>
          <XStack alignItems="center" gap="$2">
            <Input
              width={64}
              keyboardType="number-pad"
              value={String(item.quantity)}
              onChangeText={value => updateQuantity(item.id, parseInt(value, 10))}
            />
            <Button size="$2" onPress={() => removeItem(item.id)}>Remove</Button>
          </XStack>
        </XStack>
      ))}
      <Separator marginTop="$4" />
      <YStack paddingTop="$4">
        <Paragraph>Subtotal: ${(cart.subtotal / 100).toFixed(2)}</Paragraph>
        {cart.discount > 0 && (
          <Paragraph>Discount: -${(cart.discount / 100).toFixed(2)}</Paragraph>
        )}
        <Paragraph fontWeight="700">Total: ${(cart.total / 100).toFixed(2)}</Paragraph>
      </YStack>
    </YStack>
  )
}
```

## Apply Coupon

```tsx
'use client'

import { useState } from 'react'
import { useCart } from '@/components/cart-provider'
import { Button, Form, Input, Paragraph } from '@hanzo/gui'

export function CouponForm() {
  const { applyCoupon } = useCart()
  const [code, setCode] = useState('')
  const [error, setError] = useState('')

  async function handleSubmit() {
    try {
      await applyCoupon(code)
      setCode('')
      setError('')
    } catch (err: any) {
      setError(err.message || 'Invalid coupon code')
    }
  }

  return (
    <Form onSubmit={handleSubmit} gap="$2">
      <Input value={code} onChangeText={setCode} placeholder="Coupon code" />
      <Form.Trigger asChild>
        <Button>Apply</Button>
      </Form.Trigger>
      {error && (
        <Paragraph size="$2" color="$red10">
          {error}
        </Paragraph>
      )}
    </Form>
  )
}
```

<Callout type="info">
Cart state is persisted server-side. Store only the `cartId` in `localStorage` and fetch the full cart on page load.
</Callout>
